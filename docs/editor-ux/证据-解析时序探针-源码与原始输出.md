# 证据 · 解析时序探针（源码与原始输出）

对应诊断文档：`docs/editor-ux/编辑体验-位置跳动-诊断与调研.md` §1。

**结论**：`livePreviewField.update()` 的守卫（`src/editor/livePreview.ts:511`）让装饰器**永远追不上后台解析器**；`parsedTo` 那套补算机制在生产里是死代码。文档尾部会停留在「原始 markdown」，直到用户下一次点击/打字时**一次性补上 880 条装饰**，造成整篇重排。

**性质**：只读诊断，不改任何产品代码。

**为什么固化到这里**：探针原本放在 `.tmp-verify/`（该目录在 `.gitignore` 里，会被清掉）。本文件把源码与原始输出留档，保证结论可复核。

## 1. 怎么重跑

把下面 §2 的源码与 §3 的配置写到 `.tmp-verify/parse-lag/`，然后：

```bash
npx vitest run --config .tmp-verify/parse-lag/vitest.probe.config.ts
```

环境：`vitest 5.0.2`、`@codemirror/language`（`node_modules` 内的版本）。

## 2. 探针源码

文件：`.tmp-verify/parse-lag/parseLag.probe.test.ts`

```ts
/**
 * 探针（只读诊断，不改产品代码）：
 * 证明 `livePreviewField.update()` 的守卫会让装饰器**永远追不上后台解析器**。
 *
 * 被验证的实现：src/editor/livePreview.ts:507-513
 *   update(value, tr) {
 *     const settingsChanged = tr.effects.some(...)
 *     if (!tr.docChanged && !tr.selection && !settingsChanged) return value;   // ← 守卫
 *     return updateDecorations(value, tr, settingsChanged);
 *   }
 *
 * 而后台解析器推进语法树时发出的正是**只有 effect** 的事务：
 *   node_modules/@codemirror/language/dist/index.js:621
 *   this.view.dispatch({ effects: Language.setState.of(new LanguageState(field.context)) });
 *
 * 运行：npx vitest run --config .tmp-verify/parse-lag/vitest.probe.config.ts
 */
import { ensureSyntaxTree, syntaxTree } from "@codemirror/language";
import { EditorState } from "@codemirror/state";
import { describe, expect, it } from "vitest";
import { livePreviewField } from "../../src/editor/livePreview";
import { markdownSupport } from "../../src/editor/markdown";
import { editorSettings } from "../../src/editor/settings";

/** ~90 字符一个块，和 livePreview.test.ts 的 bigDoc 同构。 */
function bigDoc(blocks: number): string {
  const parts: string[] = [];
  for (let i = 0; i < blocks; i += 1) {
    parts.push(`## 标题 ${i}`, "", `一段包含 **粗体**、*斜体*、==高亮==、\`代码\` 与 $a+b$ 的正文。`, "");
    parts.push("- 列表项一", "- [ ] 任务项", "", "> 引用一行", "");
  }
  return parts.join("\n");
}

function decos(state: EditorState) {
  const out: { from: number; to: number; cls: string; widget: string }[] = [];
  for (const iter = state.field(livePreviewField).set.iter(); iter.value; iter.next()) {
    const spec = (iter.value as unknown as { spec: Record<string, unknown> }).spec;
    out.push({
      from: iter.from,
      to: iter.to,
      cls: String(spec.class ?? ""),
      widget: spec.widget ? (spec.widget as { constructor: { name: string } }).constructor.name : "",
    });
  }
  return out;
}

function stateFor(doc: string): EditorState {
  return EditorState.create({
    doc,
    extensions: [markdownSupport, editorSettings({}), livePreviewField],
  });
}

describe("探针：装饰器追不上后台解析", () => {
  it("初始化时语法树只解析到 3000 字符（Work.InitViewport），文档尾部没有装饰", () => {
    const doc = bigDoc(80); // ≈ 7000 字符
    const state = stateFor(doc);
    const tree = syntaxTree(state);
    const deco = decos(state);

    const tailDeco = deco.filter((d) => d.from > 3000);
    console.log("=== 文档长度 ===", doc.length);
    console.log("=== 初始化后语法树长度 ===", tree.length, `(${((tree.length / doc.length) * 100).toFixed(1)}%)`);
    console.log("=== 初始化后装饰总数 ===", deco.length, "其中 from>3000 的：", tailDeco.length);
    console.log("=== 初始化后行类装饰（前 5 个）===", deco.slice(0, 5));

    expect(doc.length).toBeGreaterThan(3000);
    // 树是残缺的：尾部没解析
    expect(tree.length).toBeLessThan(doc.length);
    // 尾部没有任何装饰 ⇒ 用户看到的是**原始 markdown**
    expect(tailDeco.length).toBe(0);
  });

  it("后台解析完成后，只发 effect 的事务【不会】让装饰器更新（守卫直接 return value）", () => {
    const doc = bigDoc(80);
    let state = stateFor(doc);
    const before = decos(state);

    // 1) 模拟后台 ParseWorker 把整篇解析完（推进的是共享的 ParseContext）
    ensureSyntaxTree(state, doc.length, 60_000);

    // 2) 模拟它发出的**只有 effect** 的事务。生产里这个 effect 是
    //    `Language.setState.of(new LanguageState(field.context))`
    //    （node_modules/@codemirror/language/dist/index.js:621）—— 没有 docChanged、
    //    没有 selection，effect 也不是 setEditorSettings / refreshDecorations。
    //    本探针用空 spec 复现同一形状；`Language.state` 的 field.update 会把这个事务里
    //    已经推进完的 context 树带进新 state。
    state = state.update({}).state;

    const after = decos(state);
    console.log("=== effect-only 事务前后装饰数 ===", before.length, "→", after.length);
    console.log("=== 此时语法树长度 ===", syntaxTree(state).length, "/", doc.length);

    // 语法树已经完整了 —— 但装饰器**一个都没变**
    expect(syntaxTree(state).length).toBe(doc.length);
    expect(after.length).toBe(before.length);
  });

  it("只有等到用户下一次「改选区 / 打字」事务，尾部才一次性补上装饰 ⇒ 文档当场重排", () => {
    const doc = bigDoc(80);
    let state = stateFor(doc);
    const before = decos(state);
    const beforeTail = before.filter((d) => d.from > 3000).length;

    ensureSyntaxTree(state, doc.length, 60_000);
    state = state.update({}).state; // 后台推进：装饰不动
    expect(decos(state).filter((d) => d.from > 3000).length).toBe(beforeTail);

    // 用户点了一下 / 敲了一个字 —— 这就是 livePreview.test.ts 的 settled() 干的事
    state = state.update({ selection: state.selection }).state;

    const after = decos(state);
    const afterTail = after.filter((d) => d.from > 3000).length;
    console.log("=== 补装饰前后 ===", {
      装饰总数: `${before.length} → ${after.length}`,
      尾部装饰: `${beforeTail} → ${afterTail}`,
    });
    console.log(
      "=== 一次性新增的装饰（会重排文档的部分）===",
      after.length - before.length,
      "条，其中行类：",
      after.filter((d) => d.cls.startsWith("md-h")).length,
      "条；widget：",
      after.filter((d) => d.widget).length,
      "个",
    );

    expect(afterTail).toBeGreaterThan(0);
    expect(after.length).toBeGreaterThan(before.length);
  });
});
```
